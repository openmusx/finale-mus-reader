cmake_minimum_required(VERSION 3.20)
project(reader_consumer LANGUAGES C CXX)

get_filename_component(reader_source "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(READER_CONSUMER_METHOD "SUBDIRECTORY" CACHE STRING "SUBDIRECTORY or FETCHCONTENT")
set_property(CACHE READER_CONSUMER_METHOD PROPERTY STRINGS SUBDIRECTORY FETCHCONTENT)
option(READER_CONSUMER_SUPPLY_DEPENDENCIES "Provide musxdom and zlib before the reader" ON)
if(NOT CMAKE_CXX_STANDARD)
    set(CMAKE_CXX_STANDARD 20)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(EMSCRIPTEN)
    # Exception handling must agree throughout the document construction pipeline.
    string(APPEND CMAKE_CXX_FLAGS " -fexceptions")
endif()
if(MSVC)
    add_compile_options(/bigobj /utf-8)
endif()

include("${reader_source}/cmake/ReaderDependencies.cmake")
if(READER_CONSUMER_SUPPLY_DEPENDENCIES)
    set(musx_BUILD_TESTING OFF CACHE BOOL "Do not build musxdom tests")
    set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "Do not build zlib examples")
    set(CMAKE_SUPPRESS_DEVELOPER_WARNINGS ON)
    finale_mus_reader_declare_musxdom()
    finale_mus_reader_declare_zlib()
    FetchContent_MakeAvailable(musxdom zlib)
    if(TARGET zlib)
        set_target_properties(zlib PROPERTIES EXCLUDE_FROM_ALL TRUE)
        if(EMSCRIPTEN)
            set_target_properties(zlib PROPERTIES OUTPUT_NAME z_shared)
        endif()
    endif()
    add_library(ZLIB::ZLIB ALIAS zlibstatic)
endif()

if(READER_CONSUMER_METHOD STREQUAL "SUBDIRECTORY")
    add_subdirectory("${reader_source}" reader)
elseif(READER_CONSUMER_METHOD STREQUAL "FETCHCONTENT")
    FetchContent_Declare(finale_mus_reader SOURCE_DIR "${reader_source}")
    FetchContent_MakeAvailable(finale_mus_reader)
else()
    message(FATAL_ERROR "READER_CONSUMER_METHOD must be SUBDIRECTORY or FETCHCONTENT")
endif()

if(FINALE_MUS_READER_BUILD_TESTING OR FINALE_MUS_READER_BUILD_REPORTING_TESTING
        OR TARGET finale_mus_reader_internal OR TARGET Catch2::Catch2)
    message(FATAL_ERROR "The reader enabled its tests in a consumer build")
endif()
if(TARGET pugixml OR TARGET pugixml::pugixml)
    message(FATAL_ERROR "The reader supplied an XML backend")
endif()
get_target_property(reader_definitions finale_mus_reader INTERFACE_COMPILE_DEFINITIONS)
if(reader_definitions MATCHES "FINALE_MUS_READER_ENABLE_INSTRUMENTATION|MUSX_USE_")
    message(FATAL_ERROR "The reader enabled instrumentation or an XML backend")
endif()
get_target_property(reader_dependencies finale_mus_reader LINK_LIBRARIES)
if(NOT "musx" IN_LIST reader_dependencies)
    message(FATAL_ERROR "The reader does not link the shared musx target")
endif()
if(READER_CONSUMER_SUPPLY_DEPENDENCIES)
    get_target_property(reader_zlib ZLIB::ZLIB ALIASED_TARGET)
    if(NOT reader_zlib STREQUAL "zlibstatic" OR NOT "ZLIB::ZLIB" IN_LIST reader_dependencies)
        message(FATAL_ERROR "The reader did not reuse the parent's zlib target")
    endif()
endif()

set(PUGIXML_BUILD_TESTS OFF CACHE BOOL "Do not build pugixml tests")
finale_mus_reader_declare_pugixml()
FetchContent_MakeAvailable(pugixml)
finale_mus_reader_keep_out_of_unity(pugixml pugixml-static)

set(fixture_path "${reader_source}/tests/evidence/F2012/F2012-baseline.mus")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${fixture_path}")
file(READ "${fixture_path}" fixture_hex HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," fixture_bytes "${fixture_hex}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/fixture.h"
    CONTENT "// Copyright (c) 2026 Robert G. Patterson\n// SPDX-License-Identifier: MIT\n#pragma once\nconstexpr unsigned char fixture[] = {${fixture_bytes}};\n"
    @ONLY)

add_executable(reader_consumer main.cpp)
finale_mus_reader_enable_unity(reader_consumer)
target_include_directories(reader_consumer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(reader_consumer PRIVATE MUSX_USE_PUGIXML)
target_link_libraries(reader_consumer PRIVATE finale_mus_reader::finale_mus_reader pugixml)
# The reader's public feature requirement must raise this to C++20.
set_target_properties(reader_consumer PROPERTIES CXX_STANDARD 17)

add_executable(reader_consumer_text text.cpp)
finale_mus_reader_enable_unity(reader_consumer_text)
target_include_directories(reader_consumer_text PRIVATE "${reader_source}/src")
target_link_libraries(reader_consumer_text PRIVATE finale_mus_reader::finale_mus_reader)

enable_testing()
foreach(consumer_target reader_consumer reader_consumer_text)
    if(MSVC)
        target_compile_options(${consumer_target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${consumer_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(EMSCRIPTEN)
        target_link_options(${consumer_target} PRIVATE -sFILESYSTEM=0 -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=1048576)
    endif()
    add_test(NAME ${consumer_target} COMMAND ${consumer_target})
endforeach()
